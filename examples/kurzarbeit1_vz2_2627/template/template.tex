\startluacode
local names = fonts.names
names.load()

local function font_exists(name)
  -- names.resolve akzeptiert auch einen einfachen String
  return names.resolve and names.resolve(name)
end

local main, bold

-- 1. Versuch: echtes Calibri (z.B. auf Windows)
if font_exists("calibri") or font_exists("Calibri") then
  main = "name:calibri"
  bold = "name:calibribold"

-- 2. Fallback: Carlito (NixOS-Paket)
elseif font_exists("carlito") or font_exists("Carlito") then
  main = "name:carlito"
  bold = "name:carlitobold"

-- 3. Letzte Rettung: IBM Plex Sans
else
  main = "name:ibmplexsans"
  bold = "name:ibmplexsansbold"
end

context("\\definefont[calibri]["     .. main .. " at 11pt]")
context("\\definefont[calibribold][" .. bold .. " at 14pt]")
